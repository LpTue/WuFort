import math

fi = open("ma.txt","r")
fo = open("out.txt","w")
l = fi.read().split(".")
grb = 0.5 * (1 + math.sqrt(5))

def tt(n):
    n1 = n[::-1]
    t = 0
    for i in range(0,len(n)):
        if n1[i] == "1":
            t += pow(grb,i)
    return t

# các hằng số nhiễu: không có, φ^11, φ^12, phi^13
nhieu = [0, 199, 322, 521]

kq = ""
for i in l:
    a = math.ceil(tt(str(i)) - 0.000001) 
    for f in nhieu:
        if 1 <= a - f <= 126:            
            a = a - f
            break
    kq += chr(a)

print(kq)
fo.write(kq)
